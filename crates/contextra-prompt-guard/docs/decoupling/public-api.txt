// PromptInjectionGuard
pub struct contextra_prompt_guard::PromptInjectionGuard {
    pub redaction_placeholder: String,
}
impl contextra_prompt_guard::PromptInjectionGuard {
    pub fn new(policy: QuarantinePolicy, redaction_placeholder: String, patterns: Vec<String>, audit_logger: SecurityAuditLogger) -> Self;
    pub fn with_clock(self, clock: std::sync::Arc<dyn Clock>) -> Self;
    pub fn clock(&self) -> &std::sync::Arc<dyn Clock>;
    pub fn default_patterns() -> Vec<String>;
    pub fn policy(&self) -> QuarantinePolicy;
    pub fn patterns(&self) -> &[String];
    pub fn audit_logger(&self) -> &SecurityAuditLogger;
    #[cfg(feature = "file-audit")]
    pub fn load_from_file(path: impl AsRef<std::path::Path>) -> Result<Self, GuardError>;
    #[cfg(feature = "file-audit")]
    pub fn from_env() -> Self;
    pub fn is_zero_width(c: char) -> bool;
    pub fn skeletonize_char(c: char) -> char;
    pub fn normalize_text(text: &str) -> String;
    pub fn collapse_whitespace(s: &str) -> String;
    pub fn strip_whitespace(s: &str) -> String;
    pub fn decode_base64(input: &str) -> Option<Vec<u8>>;
    pub fn extract_base64_candidates(text: &str) -> Vec<&str>;
    pub fn detect_recursive(&self, text: &str, depth: usize) -> Option<String>;
    pub fn detect(&self, text: &str) -> Option<String>;
    pub fn process_result(&self, doc_id: &str, collection: &str, obj: &mut serde_json::Map<String, serde_json::Value>) -> bool;
}
